#### код для модели(?):
```python
import tensorflow as tf

  

# загрузка датасета MNIST

mnist = tf.keras.datasets.mnist

(x_train, y_train), (x_test, y_test) = mnist.load_data()

  

# нормализация данных (перевод пикселей из диапазона 0–255 в диапазон от 0 до 1)

# чтобы нейросеть обучалась быстрее и стабильнее

x_train, x_test = x_train / 255.0, x_test / 255.0

  

# Сборка архитектуры нейросети

model = tf.keras.models.Sequential([

    # превращае квадратной матрицы картинки 28x28 в плоский вектор из 784 пикселей

    tf.keras.layers.Flatten(input_shape=(28, 28)),

    # Скрытый слой из 128 нейронов, активируемый функцией ReLU

    tf.keras.layers.Dense(128, activation='relu'),

    # Dropout временно отключает 20% случайных нейронов, чтобы сеть не затачивалась под датасет, а искала закономерности

    tf.keras.layers.Dropout(0.2),                  

    # Выходной слой - 10 нейронов (по одному на каждую цифру от 0 до 9)

    tf.keras.layers.Dense(10, activation='softmax')

])

  

# настройка процесса обучения (выбор оптимизатора и функции потерь)

model.compile(optimizer='adam',

              loss='sparse_categorical_crossentropy',

              metrics=['accuracy'])

  

# Обучение нейросети (5 циклов-эпох прохождения по всем картинкам)

print("Начался процесс обучения...")

model.fit(x_train, y_train, epochs=5)

  

# проверяем, как сеть справляется с картинками, которые она никогда не видела

print("\nПроверка на тестовых данных:")

test_loss, test_acc = model.evaluate(x_test, y_test, verbose=2)

  

print(f"\nТочность распознавания: {test_acc * 100:.2f}%")
```
#### код для подгрузки своих картинок цифр:
картинка должна называться my_digit.png // я рисовал в paint квадратные цифры, жирным шрифтом чтобы при обработке не терялась линия
```python
from PIL import Image, ImageOps

import numpy as np

  

# открытие обычной картинку

img = Image.open('my_digit.png')

  

# переводи в оттенки серого и сжатие до стандартных 28x28 пикселей

img = img.convert('L').resize((28, 28))

  

# Инвертация цвета до белого

img = ImageOps.invert(img)

  

# перевод картинки в массив чисел и нормализация (деление на 255)

img_array = np.array(img) / 255.0

  

# модель ожидает на вход пачку картинок, поэтому форма (28, 28) превращается в форму (1, 28, 28)

img_array = np.expand_dims(img_array, axis=0)

  

prediction = model.predict(img_array)

  

predicted_digit = np.argmax(prediction)

  

print(f"\nНейросеть считает, что это цифра: {predicted_digit}")
```
### Поиск кратчайшего пути в графе
Код реализует **волновой алгоритм (алгоритм Ли)** для поиска кратчайшего пути в двумерном лабиринте с препятствиями. Сначала он запускает поиск в ширину (BFS) с помощью двусторонней очереди (`deque`): равномерно «*распространяет волну*» из стартовой точки по всем свободным клеткам, записывая в них количество шагов от начала. Как только волна достигает финиша, программа включает обратный ход — отслеживает путь назад к старту, просто переходя по клеткам со строго убывающими числами. В конце код выводит в консоль числовую карту волны, список координат кратчайшего маршрута и наглядную визуализацию, где сам путь нарисован звездочками.
